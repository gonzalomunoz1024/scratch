apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: inspect-original-data
  namespace: opensearch
spec:
  accessModes:
  - ReadWriteOnce
  storageClassName: sc-ontap-nas
  resources:
    requests:
      storage: 20Gi
  volumeName: pvc-d6c8f5a4-8370-45bc-9848-5198049e296f
---
apiVersion: batch/v1
kind: Job
metadata:
  name: data-inspector
  namespace: opensearch
spec:
  ttlSecondsAfterFinished: 3600
  backoffLimit: 0
  template:
    spec:
      serviceAccountName: opensearch-sa
      restartPolicy: Never
      containers:
      - name: inspector
        image: docker.io/busybox:1.36
        command:
        - sh
        - -c
        - |
          echo "=== Top-level contents ==="
          ls -la /data
          echo ""
          echo "=== Total size ==="
          du -sh /data
          echo ""
          echo "=== Sizes by top-level dir ==="
          du -sh /data/* 2>/dev/null
          echo ""
          echo "=== Indices (if any) ==="
          ls -la /data/nodes/0/indices/ 2>&1 || echo "indices/ not at default path"
          echo ""
          echo "=== Directory tree (depth 5) ==="
          find /data -maxdepth 5 -type d 2>&1 | head -60
          echo ""
          echo "=== State files ==="
          find /data -name "node-*.st" -o -name "manifest-*.st" 2>/dev/null | head -10
          echo ""
          echo "=== Index UUIDs ==="
          ls /data/nodes/0/indices/ 2>/dev/null | head -20
          echo "=== DONE ==="
        volumeMounts:
        - name: data
          mountPath: /data
          readOnly: true
        securityContext:
          allowPrivilegeEscalation: false
          runAsNonRoot: false
          capabilities:
            drop: ["ALL"]
        resources:
          requests:
            cpu: 100m
            memory: 128Mi
          limits:
            cpu: 500m
            memory: 512Mi
      volumes:
      - name: data
        persistentVolumeClaim:
          claimName: inspect-original-data


oc delete pod data-inspector -n opensearch --ignore-not-found
oc delete pvc inspect-original-data -n opensearch --ignore-not-found

oc get pv pvc-d6c8f5a4-8370-45bc-9848-5198049e296f
